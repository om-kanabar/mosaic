def _combine(shift, i, j):
    # Determine how far to delay the start of each signal relative to a global timeline (t=0)
    delay_i = abs(shift) if shift < 0 else 0
    delay_j = shift if shift > 0 else 0
    
    total_len = max(length(i) + delay_i, length(j) + delay_j)
    
    # Calculate overlap bounds strictly where both signals have valid data
    overlap_start = max(delay_i, delay_j)
    overlap_end = min(delay_i + length(i), delay_j + length(j))
    overlap_length = overlap_end - overlap_start
    
    x = []
    
    for t from 0 to total_len - 1:
        # Map the global time 't' back to the local index of each signal
        idx_i = t - delay_i
        idx_j = t - delay_j
        
        in_i = (0 <= idx_i < length(i))
        in_j = (0 <= idx_j < length(j))
        
        if in_i and in_j:
            if overlap_length > 0:
                weight_j = (t - overlap_start) / overlap_length
            else:
                weight_j = 0.5
                
            weight_i = 1.0 - weight_j
            blended_val = (i[idx_i] * weight_i) + (j[idx_j] * weight_j)
            x.append(blended_val)
            
        elif in_i:
            x.append(i[idx_i])
        elif in_j:
            x.append(j[idx_j])
    
    CSV.save(x)
    return x
