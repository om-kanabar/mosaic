def combine(cross_point, i, j):
    # 1. Align the signals using the offset (same as before)
    if cross_point >= 0:
        aligned_i = i
        aligned_j = pad_beginning(j, amount=cross_point)
    else:
        aligned_i = pad_beginning(i, amount=abs(cross_point))
        aligned_j = j
        
    max_len = max(length(aligned_i), length(aligned_j))
    aligned_i = pad_end(aligned_i, to_length=max_len)
    aligned_j = pad_end(aligned_j, to_length=max_len)
    
    overlap_start = max(0, cross_point if cross_point >= 0 else 0)
    overlap_end = min(length(i), length(j) + (cross_point if cross_point >= 0 else 0))
    overlap_length = overlap_end - overlap_start
    
    x = []
    
    for index from 0 to max_len - 1:
        val_i = aligned_i[index]
        val_j = aligned_j[index]
        
        if val_i exists and val_j exists:
            weight_j = (index - overlap_start) / overlap_length
            weight_i = 1.0 - weight_j
            
            blended_val = (val_i * weight_i) + (val_j * weight_j)
            x.append(blended_val)
        elif val_i exists:
            x.append(val_i)
        else:
            x.append(val_j)
            
    CSV.Save(x) 
    return x
