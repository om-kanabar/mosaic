def _combine(shift, i, j):
    
    # Calculate exactly where the overlap occurs
    if shift >= 0:
        overlap_start_i = shift
        overlap_start_j = 0
    else:
        overlap_start_i = 0
        overlap_start_j = abs(shift)
        
    # Determine the total length of the combined signal
    # to the end of the second signal
    total_len = max(length(i), length(j) + shift) if shift >= 0 else max(length(j), length(i) + abs(shift))
    
    # Calculate how long the overlapping region actually is
    overlap_length = min(length(i) - overlap_start_i, length(j) - overlap_start_j)
    
    x = []
    
    for idx from 0 to total_len - 1:
        # Determine the local index for each individual signal
        idx_i = idx
        idx_j = idx - shift
        
        # Check if the current index falls within BOTH signals (Overlap / Blending Zone)
        if (0 <= idx_i < length(i)) and (0 <= idx_j < length(j)):
            # Linear fade: weight_j goes from 0 to 1 across the overlap length
            weight_j = (idx_i - overlap_start_i) / overlap_length
            weight_i = 1.0 - weight_j
            
            blended_val = (i[idx_i] * weight_i) + (j[idx_j] * weight_j)
            x.append(blended_val)
            
        # Check if the index only exists in signal i
        elif 0 <= idx_i < length(i):
            x.append(i[idx_i])
            
        # Check if the index only exists in signal j
        elif 0 <= idx_j < length(j):
            x.append(j[idx_j])

    CSV.Save(x) 
    return x

def _synthesizeSample(D, U)
    selected_samples = _select_samples(D, U)
    selected_samples = _initial_comparison(D, U, *selected_samples)

    z = _combine(shift, *selected_samples)
